三、kanzi代码

3.1 onConfigure

```
virtual void onConfigure(ApplicationProperties& configuration) KZ_OVERRIDE
{
HWND desktop = GetDesktopWindow(); // 获取桌面窗口句柄
RECT rect;
GetWindowRect(desktop, &rect); // 获取桌面窗口的矩形区域

screenWidth = rect.right - rect.left; // 右边界 - 左边界
screenHeight = rect.bottom - rect.top; // 下边界 - 上边界

kzLogDebug(( "Screen Width: {}", screenWidth));
kzLogDebug(( "Screen Height: {}", screenHeight));

configuration.binaryName = "city.kzb.cfg";
configuration.defaultWindowProperties.width = screenWidth;
configuration.defaultWindowProperties.height = screenHeight;
}

```

onConfigure 是运行kanzi时候的配置函数,这里我们设置全屏显示

3.2 onProjectLoaded

```
virtual void onProjectLoaded() KZ_OVERRIDE
{
// Project file has been loaded from .kzb file.

// Add initialization code here.
ShowCursor(FALSE);

// 获取屏幕中心坐标
mCenterX = screenWidth / 2;
mCenterY = screenHeight / 2;

// 将鼠标光标移动到屏幕中心
SetCursorPos(mCenterX, mCenterY);

player = getScreen()->lookupNode<Node3D>("#player");
if (player == nullptr) {
kzLogDebug(("player no find"));
exit(0);
}

camera = player->lookupNode<Camera>("Camera");
if (camera == nullptr) {
kzLogDebug(("camera no find"));
exit(0);
}

playerMesh = player->lookupNode<Model3D>("body");
if (playerMesh == nullptr) {
kzLogDebug(("playerMesh no find"));
exit(0);
}

city = getScreen()->lookupNode<Node3D>("#city");
if (city == nullptr) {
kzLogDebug(("city no find"));
exit(0);
}
}

```

onProjectLoaded是kanzi加载kzb界面后执行的函数,这里我们提前获取一些节点。并隐藏鼠标,设置鼠标到屏幕中心。

3.6 updatePlayer

```
void updatePlayer(chrono::milliseconds deltaTime) {

{
POINT p;
GetCursorPos(&p);

// 计算鼠标移动的偏移量
int offsetX = p.x - mCenterX;
int offsetY = p.y - mCenterY;

// 更新玩家视角
updatePlayerView(offsetX, offsetY, deltaTime);

// 将鼠标光标重新移动到屏幕中心
SetCursorPos(mCenterX, mCenterY);
}

```

首先计算鼠标偏移,并把鼠标重置回屏幕中心

KzcVector3 rotatedFront;
KzcVector3 rotatedRight;

3.7 updatePlayerView

```
void updatePlayerView(int offsetX, int offsetY, chrono::milliseconds deltaTime)
{
// 根据鼠标移动的偏移量更新玩家的视角
// 这里只是一个示例,你需要根据你的游戏逻辑来实现具体的视角更新
float sensitivity = 0.00005f; // 鼠标灵敏度
mYaw -= offsetX * sensitivity * deltaTime.count();
mPitch -= offsetY * sensitivity * deltaTime.count();

// 限制俯仰角
if (mPitch > 1.0f) mPitch = 1.0f;
if (mPitch < 0.0f) mPitch = 0.0f;

// 更新玩家的旋转
updatePlayerRotation();
}

```

updatePlayerView计算俯仰角和偏航角,并对边界判断

3.7 updatePlayerRotation

```
void updatePlayerRotation()
{
SRTValue3D _SRTValue3DPlayer = player->getRenderTransformation();
KzcQuaternion _rotationPlayer = _SRTValue3DPlayer.getRotation();

SRTValue3D _SRTValue3DCamera = camera->getRenderTransformation();
KzcQuaternion _rotationCamera = _SRTValue3DCamera.getRotation();

// Update player rotation
KzcQuaternion yawRotation = kzcQuaternionFromRotationAngles(0.0, mYaw, 0.0);
_rotationPlayer = kzcQuaternionCreateNormalized(yawRotation);
_SRTValue3DPlayer.setRotation(_rotationPlayer);
player->setRenderTransformation(_SRTValue3DPlayer);

// Update camera rotation
KzcQuaternion pitchRotation = kzcQuaternionFromRotationAngles(mPitch, 0.0, 0.0);
_rotationCamera = kzcQuaternionCreateNormalized(pitchRotation);
_SRTValue3DCamera.setRotation(_rotationCamera);
camera->setRenderTransformation(_SRTValue3DCamera);
}

```

updatePlayerRotation根据鼠标移动,分别旋转玩家和摄像机
