# kanzi多语言优化

> 原文: https://blog.csdn.net/chen_227/article/details/156143820  
> 发布: 2025-12-22  

背景:kanzi开发HMI界面时候,需要实现多语言。越复杂的界面,使用的文本节点越多,切换多语言时候,就越卡顿。本方案提供优化方案,保证切换语言流畅,画面不掉帧。

一、优化字体文件

除了多媒体文字,其他文字基本都是固定的,通过裁剪ttf字体文件,减少字体文件大小,减少切换语言时候查找字体时间。比如一个字体文件里面有3万字,从3万字里面查找需要显示的字符,肯定没有从300字里面查找快。

1. 准备字体文件,可以是ttf,otf

2. 收集所有固定显示文言,可以从table表里面收集

打开主工程,右键Localization Table,Export,得到\Localization\Localization Table

3. 编写脚本,提取文字

运行merge_po.bat,得到merged_output.txt,这是所有语言的所有文字
merge_po.bat 如下

```
&#64;echo off
setlocal enabledelayedexpansion

:: 设置目标目录和输出文件
set TARGET_DIR=Localization\Localization Table
set OUTPUT_FILE=merged_output.txt

:: 检查目标目录是否存在
if not exist "%TARGET_DIR%" (
echo error: dir no exist: %TARGET_DIR%
pause
exit /b 1
)

:: 清空输出文件&#xff08;如果存在&#xff09;
if exist "%OUTPUT_FILE%" del "%OUTPUT_FILE%"

:: 遍历目标目录下的所有 .po 和 .pot 文件
for %%f in ("%TARGET_DIR%\*.po" "%TARGET_DIR%\*.pot") do (
echo handle: %%~nxf
type "%%f" >> "%OUTPUT_FILE%"
)

echo.
echo finish "%OUTPUT_FILE%"
pause

```

fontmin_batch.py 如下

```
#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""
批量字体子集化工具&#xff08;支持 ttf/otf/woff/woff2&#xff09;
Usage: python subsetter.py
"""

import os
import re
from fontTools.ttLib import TTFont
from fontTools.subset import Subsetter
from fontTools.subset import Options
import shutil

# ========== 用户只需改这里 ==========
INPUT_TXT = "merged_output.txt" # 一行文本,包含所有需要的字符
INPUT_DIR = "fonts" # 原始字体目录
OUTPUT_DIR = "output" # 裁剪后输出目录
# ====================================

SUPPORT_EXTS = {".ttf", ".otf", ".woff", ".woff2"}

def read_chars(txt_file):
"""读取文本并去重、去空白"""
if not os.path.isfile(txt_file):
raise FileNotFoundError(f"字符文件不存在: {txt_file}")
with open(txt_file, encoding="utf-8") as f:
text = f.read()
chars = "".join(sorted(set(re.sub(r"\s&#43;", "", text))))
print(f"待保留字符数&#xff08;去重后&#xff09;: {len(chars)}")
return chars

def make_subset(font_path, chars, output_path):
try:
font = TTFont(font_path)

# 1. 构造选项
opt = Options()
opt.drop_tables &#43;= ["GDEF", "GSUB", "GPOS", "MATH", "kern"]
opt.hinting = False
opt.notdef_glyph = False
opt.layout_features = []
opt.layout_closure = False
opt.ignore_missing_glyphs = True
opt.ignore_missing_unicodes = True

# 2. 子集化
subsetter = Subsetter(options=opt)
subsetter.populate(text=chars)
subsetter.subset(font)
font.save(output_path)
print(f"✅ 成功裁剪: {os.path.basename(font_path)}")
except Exception as e:
# 3. 任何失败都直接复制
shutil.copy2(font_path, output_path)
print(f"⚠️ 裁剪失败&#xff08;已直接复制&#xff09;: {os.path.basename(font_path)} - {e}")

def main():
chars = read_chars(INPUT_TXT)
if not chars:
print("字符表为空,程序终止。")
return

os.makedirs(OUTPUT_DIR, exist_ok=True)

files = [f for f in os.listdir(INPUT_DIR)
if os.path.splitext(f.lower())[1] in SUPPORT_EXTS]
if not files:
print(f"在 {INPUT_DIR} 里未找到任何字体文件。")
return

print(f"开始处理 {len(files)} 个字体…\n")
for name in files:
in_path = os.path.join(INPUT_DIR, name)
out_path = os.path.join(OUTPUT_DIR, name)
try:
make_subset(in_path, chars, out_path)
except Exception as e:
print(f"❌ 彻底失败: {name} -> {e}")

print("\n===== 全部任务完成 =====")

if __name__ == "__main__":
main()

```

kanzi通过resource id来自动更新Localization表,虽然方便,但是不智能,我们希望通过插件实现定制化更新,只更新显示的文字的多语言。

2. renderOverride

原理就是通过继承renderOverride函数,实现文本节点渲染重载,在这个函数里面,检查节点是否visible和opacity>0

```
void PluginTextBlock2DLanguage::renderOverride(Renderer3D& renderer,
CompositionStack& compositionStack/*, const optional<Matrix3x3>& baseTransform*/) {
TextBlock2D::renderOverride(renderer, compositionStack/*, baseTransform*/);

//visible, opacity>0.0
string rootPath = getProperty(RootPageRefProperty);
NodeSharedPtr node = lookupNode<Node>(rootPath);
if (node == nullptr) {
kzLogDebug(("{} is not find", rootPath));
return;
}

Screen* screen = dynamic_cast<Screen*>(node->getParent());
if (screen == nullptr) {
kzLogDebug(("screen is null"));
return;
}

if (getProperty(Node::VisibleProperty) == false || getProperty(Node::OpacityProperty) == 0.0){
return;
}

string local = screen->getLocale();

if (local != m_localStr) { 
kzLogDebug(("local, {} change to {}", m_localStr, local));
m_localStr = local;

handleSearchDict();
}

}

```
