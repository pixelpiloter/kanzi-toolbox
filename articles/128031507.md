// x [-5.0~5.0]
float gaussian(float x, float mean, float stddev) {
const float pi = 3.14159265358979323846; // 定义pi常量
// 计算高斯分布的概率密度函数

}

vec2 DropLayer2(vec2 uv, float t) {

}

return sum;
}
